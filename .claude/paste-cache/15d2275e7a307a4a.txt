+SnxTransTcpRecvN (UINT1 *pu1Buf, UINT4 u4Want)
+{
+    UINT4               u4Got = 0;
+    INT4                i4Ret;
+    struct pollfd       PollFd;
+
+    while (u4Got < u4Want)
+    {
+        i4Ret = recv (gAgtxGlobalInfo.i4SockId, pu1Buf + u4Got,
+                      (INT4) (u4Want - u4Got), 0);
+        if (i4Ret > 0)
+        {
+            u4Got += (UINT4) i4Ret;
+            continue;
+        }
+        if (i4Ret == 0)
+        {
+            /* Orderly shutdown by the master */
+            return SNMP_FAILURE;
+        }
+        if (errno == EINTR)
+        {
+            continue;
+        }
+        if ((errno == EAGAIN) || (errno == EWOULDBLOCK))
+        {
+            PollFd.fd = gAgtxGlobalInfo.i4SockId;
+            PollFd.events = POLLIN;
+            PollFd.revents = 0;
+            if (poll (&PollFd, 1, SNX_TRANS_POLL_TIMEOUT_MS) <= 0)
+            {
+                return SNMP_FAILURE;
+            }
+            continue;
+        }
+        return SNMP_FAILURE;
+    }
+    return SNMP_SUCCESS;
